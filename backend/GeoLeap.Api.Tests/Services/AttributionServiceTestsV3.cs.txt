    [Fact]
    public async Task CalculateAttributionAsync_WithValidConversionEvent_ReturnsAttribution()
    {
        // Arrange
        var conversionEventId = Guid.NewGuid();
        var expectedAttribution = new AttributionResult
        {
            ConversionEventId = conversionEventId,
            TotalAttribution = 1.0m,
            AttributionModelId = Guid.NewGuid(),
            ModelName = "Last Touch",
            Touchpoints = new List<AttributionTouch>
            {
                CreateValidAttributionTouch("organic_search", 0.6m),
                CreateValidAttributionTouch("paid_search", 0.4m)
            }
        };

        _attributionService.CalculateAttributionAsync(Arg.Any<Guid>(), Arg.Any<Guid?>())
            .Returns(Task.FromResult(expectedAttribution));

        // Act
        var result = await _attributionService.CalculateAttributionAsync(conversionEventId);

        // Assert
        Assert.NotNull(result);
        Assert.Equal(conversionEventId, result.ConversionEventId);
        Assert.Equal(1.0m, result.TotalAttribution);
        Assert.Equal(2, result.Touchpoints.Count);
    }

    [Fact]
    public async Task CompareAttributionModelsAsync_WithMultipleModels_ReturnsComparison()
    {
        // Arrange
        var startDate = new DateTime(2023, 1, 1);
        var endDate = new DateTime(2023, 1, 31);
        var modelIds = new[] { Guid.NewGuid(), Guid.NewGuid() };
        
        var expectedComparison = new AttributionModelComparisonResult
        {
            StartDate = startDate,
            EndDate = endDate,
            Models = new List<ModelComparisonSummary>
            {
                new()
                {
                    ModelId = modelIds[0],
                    ModelName = "First Touch",
                    ModelType = AttributionModelType.FirstTouch,
                    TotalConversions = 100,
                    TotalAttributedValue = 50000m,
                    TopChannel = "organic_search",
                    TopChannelPercentage = 60.0m
                },
                new()
                {
                    ModelId = modelIds[1],
                    ModelName = "Last Touch",
                    ModelType = AttributionModelType.LastTouch,
                    TotalConversions = 100,
                    TotalAttributedValue = 50000m,
                    TopChannel = "paid_search",
                    TopChannelPercentage = 55.0m
                }
            },
            ChannelComparisons = new List<ChannelComparisonResult>
            {
                new()
                {
                    Channel = "organic_search",
                    ModelResults = new List<ChannelModelResult>
                    {
                        new() { ModelId = modelIds[0], ModelName = "First Touch", AttributedValue = 30000m, AttributedPercentage = 60.0m, Rank = 1 },
                        new() { ModelId = modelIds[1], ModelName = "Last Touch", AttributedValue = 20000m, AttributedPercentage = 40.0m, Rank = 2 }
                    },
                    VariancePercentage = 20.0m,
                    StandardDeviation = 7071.07m
                }
            }
        };

        _attributionService.CompareAttributionModelsAsync(Arg.Any<DateTime>(), Arg.Any<DateTime>(), Arg.Any<IEnumerable<Guid>>())
            .Returns(Task.FromResult(expectedComparison));

        // Act
        var result = await _attributionService.CompareAttributionModelsAsync(startDate, endDate, modelIds);

        // Assert
        Assert.NotNull(result);
        Assert.Equal(startDate, result.StartDate);
        Assert.Equal(endDate, result.EndDate);
        Assert.Equal(2, result.Models.Count);
        Assert.Single(result.ChannelComparisons);
        Assert.All(result.Models, m => Assert.Equal(100, m.TotalConversions));
    }

    [Fact]

    {
        // Arrange
        var startDate = new DateTime(2023, 1, 1);
        var endDate = new DateTime(2023, 1, 31);
        var expectedPerformance = new List<ChannelPerformanceResult>
        {
            new()
            {
                Channel = "organic_search",
                Source = "google",
                Medium = "organic",
                Campaign = "",
                Impressions = 10000,
                Clicks = 500,
                Conversions = 25,
                AttributedValue = 12500m,
                ClickThroughRate = 5.0m,
                ConversionRate = 5.0m,
                CostPerClick = 0m,
                CostPerConversion = 0m,
                ReturnOnAdSpend = 0m,
                AverageTimeToConversion = TimeSpan.FromDays(3)
            },
            new()
            {
                Channel = "paid_search",
                Source = "google",
                Medium = "cpc",
                Campaign = "brand_campaign",
                Impressions = 5000,
                Clicks = 250,
                Conversions = 20,
                AttributedValue = 10000m,
                ClickThroughRate = 5.0m,
                ConversionRate = 8.0m,
                CostPerClick = 2.0m,
                CostPerConversion = 25.0m,
                ReturnOnAdSpend = 20.0m,
                AverageTimeToConversion = TimeSpan.FromDays(1)
            }
        };

        _attributionService.GetChannelPerformanceAsync(Arg.Any<DateTime>(), Arg.Any<DateTime>(), Arg.Any<Guid?>())
            .Returns(Task.FromResult<IEnumerable<ChannelPerformanceResult>>(expectedPerformance));

        // Act
        var result = await _attributionService.GetChannelPerformanceAsync(startDate, endDate);

        // Assert
        Assert.NotNull(result);
        var performanceList = result.ToList();
        Assert.Equal(2, performanceList.Count);
        
        var organicSearch = performanceList.First(p => p.Channel == "organic_search");
        Assert.Equal(25, organicSearch.Conversions);
        Assert.Equal(12500m, organicSearch.AttributedValue);
        Assert.Equal(5.0m, organicSearch.ClickThroughRate);
        
        var paidSearch = performanceList.First(p => p.Channel == "paid_search");
        Assert.Equal(20, paidSearch.Conversions);
        Assert.Equal(10000m, paidSearch.AttributedValue);
        Assert.Equal(20.0m, paidSearch.ReturnOnAdSpend);
    }

    [Fact]
    public async Task GetChannelPerformanceAsync_ReturnsAccurateMetrics()
    {
        // Arrange
        var startDate = DateTime.UtcNow.AddDays(-30);
        var endDate = DateTime.UtcNow;
        
        var highPerformanceChannel = new ChannelPerformanceResult
        {
            Channel = "email",
            Conversions = 100,
            AttributedValue = 50000m,
            ConversionRate = 10.0m,
            ReturnOnAdSpend = 25.0m
        };

        _attributionService.GetChannelPerformanceAsync(Arg.Any<DateTime>(), Arg.Any<DateTime>(), Arg.Any<Guid?>())
            .Returns(Task.FromResult<IEnumerable<ChannelPerformanceResult>>(new[] { highPerformanceChannel }));

        // Act
        var result = await _attributionService.GetChannelPerformanceAsync(startDate, endDate);

        // Assert
        Assert.NotNull(result);
        var channel = result.First();
        Assert.True(channel.ConversionRate > 5.0m); // High conversion rate
        Assert.True(channel.ReturnOnAdSpend > 10.0m); // Good ROAS
        Assert.True(channel.AttributedValue > 0); // Revenue attribution
    }

    private static AttributionTouch CreateValidAttributionTouch(string channel, decimal attribution, DateTime? timestamp = null)
    {
        return new AttributionTouch
        {
            Id = Guid.NewGuid(),
            EventName = "touchpoint",
            Channel = channel,
            Source = $"{channel}_source",
            Medium = $"{channel}_medium",
            Campaign = $"{channel}_campaign",
            Timestamp = timestamp ?? DateTime.UtcNow.AddDays(-5),
            Attribution = attribution,
            Properties = "{}",
            Position = 1,
            PositionFromEnd = 1
        };
    }

    private static AttributionModel CreateAttributionModel(string name, AttributionModelType type, bool isDefault)
    {
        return new AttributionModel
        {
            Id = Guid.NewGuid(),
            Name = name,
            Type = type,
            Description = $"Test {name} model",
            Settings = "{}",
            IsDefault = isDefault,
            CreatedAt = DateTime.UtcNow,
            UpdatedAt = DateTime.UtcNow
        };
    }
