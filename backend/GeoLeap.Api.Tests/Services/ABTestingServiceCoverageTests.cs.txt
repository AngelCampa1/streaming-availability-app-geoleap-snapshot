    #region Integration Scenarios

    [Fact]
    public async Task FullTestLifecycle_ExecutesAllPaths()
    {
        // Arrange & Act
        // 1. Create test
        var testId = await CreateTestABTest();
        Assert.NotEmpty(testId);

        // 2. Start test
        Assert.True(await _service.StartABTestAsync(testId));

        // 3. Assign users
        await SetupActiveTestsCache(new List<ABTest> { CreateABTestModel(testId, ABTestStatus.Active) });
        var assignment1 = await _service.AssignUserToTestAsync("user1", "session1");
        var assignment2 = await _service.AssignUserToTestAsync("user2", "session2");
        Assert.NotNull(assignment1);
        Assert.NotNull(assignment2);

        // 4. Record results
        await _service.RecordABTestResultAsync(new Models.ABTestResult
        {
            TestId = testId,
            UserId = "user1",
            VariantId = assignment1.VariantId,
            MetricName = "conversion",
            MetricValue = 1.0,
            RecordedAt = DateTime.UtcNow
        });

        // 5. Pause test
        Assert.True(await _service.PauseABTestAsync(testId));

        // 6. Resume test
        Assert.True(await _service.StartABTestAsync(testId));

        // 7. Complete test
        Assert.True(await _service.CompleteABTestAsync(testId));

        // 8. Delete test
        Assert.True(await _service.DeleteABTestAsync(testId));
    }

    [Fact]
    public async Task MultipleUserAssignments_MaintainsConsistency()
    {
        // Arrange
        var testId = await CreateTestABTest();
        await _service.StartABTestAsync(testId);
        await SetupActiveTestsCache(new List<ABTest> { CreateABTestModel(testId, ABTestStatus.Active) });

        // Act - Assign same user multiple times
        var assignment1 = await _service.AssignUserToTestAsync("consistent-user", "session1");
        var assignment2 = await _service.AssignUserToTestAsync("consistent-user", "session2");
        var assignment3 = await _service.AssignUserToTestAsync("consistent-user", "session3");

        // Assert - All assignments should be consistent
        Assert.Equal(assignment1.VariantId, assignment2.VariantId);
        Assert.Equal(assignment2.VariantId, assignment3.VariantId);
    }

    #region Helper Methods

    private async Task<string> CreateTestABTest(int trafficPercentage = 100)
    {
        var request = new CreateABTestRequest
        {
            Name = $"Test {Guid.NewGuid()}",
            Description = "Test Description",
            TrafficPercentage = trafficPercentage,
            Variants = new List<CreateABTestVariantRequest>
            {
                new() { Name = "Control", TrafficWeight = 50, RankingConfiguration = new RankingConfiguration() },
                new() { Name = "Variant A", TrafficWeight = 50, RankingConfiguration = new RankingConfiguration() }
            }
        };

        return await _service.CreateABTestAsync(request, "test-user");
    }

    private ABTest CreateABTestModel(string testId, ABTestStatus status, int trafficPercentage = 100)
    {
        return new ABTest
        {
            TestId = testId,
            Name = $"Test {testId}",
            IsActive = status == ABTestStatus.Active,
            Status = status,
            TrafficPercentage = trafficPercentage,
            StartDate = DateTime.UtcNow.AddDays(-1),
            EndDate = DateTime.UtcNow.AddDays(30),
            Variants = new List<ABTestVariant>
            {
                new() { VariantId = "control", Name = "Control", TrafficWeight = 50, RankingConfiguration = new RankingConfiguration() },
                new() { VariantId = "test", Name = "Test", TrafficWeight = 50, RankingConfiguration = new RankingConfiguration() }
            }
        };
    }

    private async Task SetupActiveTestsCache(List<ABTest> tests)
    {
        var cacheKey = "active_abtests";
        var json = JsonSerializer.Serialize(tests);
        var jsonBytes = Encoding.UTF8.GetBytes(json);

        await _cache.SetAsync(cacheKey, jsonBytes, new DistributedCacheEntryOptions
        {
            AbsoluteExpirationRelativeToNow = TimeSpan.FromMinutes(10)
        });
    }

    private async Task SetupMetricsCache(string testId, ABTestMetrics metrics)
    {
        var cacheKey = $"metrics:{testId}";
        var json = JsonSerializer.Serialize(metrics);
        var jsonBytes = Encoding.UTF8.GetBytes(json);

        await _cache.SetAsync(cacheKey, jsonBytes, new DistributedCacheEntryOptions
        {
            AbsoluteExpirationRelativeToNow = TimeSpan.FromMinutes(10)
        });
    }

    #endregion
