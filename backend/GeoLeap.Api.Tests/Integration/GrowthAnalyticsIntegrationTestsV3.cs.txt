        var sessionId = "integration-session-001";

        var pageViewEvent = CreateEventRequest("page_view", "engagement", userId, sessionId);
        
        // Step 2: Track button click (consideration)
        var clickEvent = CreateEventRequest("button_click", "interaction", userId, sessionId);
        
        // Step 3: Track purchase (conversion)
        var purchaseEvent = CreateEventRequest("purchase", "conversion", userId, sessionId, 99.99m);

        // Mock the tracking to return success
        _mockTrackingService.TrackEventAsync(Arg.Any<GrowthEvent>(), Arg.Any<CancellationToken>())

            ConversionEventId = Guid.NewGuid(),

            Touchpoints = new List<AttributionTouch>

                new() { Channel = "organic_search", Attribution = 0.4m },
                new() { Channel = "email", Attribution = 0.3m },
                new() { Channel = "direct", Attribution = 0.3m }
            }
        };
        _mockAttributionService.CalculateAttributionAsync(Arg.Any<Guid>(), Arg.Any<Guid?>())
            .Returns(Task.FromResult(mockAttribution));

        // Act & Assert - Track each event in the funnel

        Assert.True(IsSuccessStatusCode(pageViewResponse.StatusCode));

        var clickResponse = await PostEventAsync(clickEvent);
        Assert.True(IsSuccessStatusCode(clickResponse.StatusCode));

        var purchaseResponse = await PostEventAsync(purchaseEvent);
        Assert.True(IsSuccessStatusCode(purchaseResponse.StatusCode));

        // Get attribution for the purchase

        var attributionResponse = await _client.GetAsync($"/api/GrowthAnalytics/attribution/{conversionEventId}");

    [Fact]
    public async Task ConcurrentRequests_MultipleEndpoints_HandlesLoad()
    {
        // Arrange - Concurrent request load test
        var tasks = new List<Task<HttpResponseMessage>>();

        // Create multiple concurrent requests to different endpoints
        for (int i = 0; i < 10; i++)
        {
            var eventRequest = CreateEventRequest($"concurrent_event_{i}", "load_test", $"user_{i}", $"session_{i}");
            tasks.Add(PostEventAsync(eventRequest));
        }

        for (int i = 0; i < 5; i++)
        {
            tasks.Add(_client.GetAsync("/api/GrowthAnalytics/stats"));
        }

        for (int i = 0; i < 3; i++)
        {
            var startDate = DateTime.UtcNow.AddDays(-30);
            var endDate = DateTime.UtcNow;
            tasks.Add(_client.GetAsync($"/api/GrowthAnalytics/attribution/summary?startDate={startDate:yyyy-MM-dd}&endDate={endDate:yyyy-MM-dd}"));
        }

        // Act
        var responses = await Task.WhenAll(tasks);

        // Assert - All requests should complete successfully or with expected errors
        Assert.All(responses, response =>
        {
            Assert.True(IsSuccessStatusCode(response.StatusCode) || 
                       IsExpectedErrorCode(response.StatusCode));
        });
    }

    private async Task<HttpResponseMessage> PostEventAsync(GrowthEventRequest eventRequest)
    {
        var json = JsonConvert.SerializeObject(eventRequest);
        var content = new StringContent(json, Encoding.UTF8, "application/json");
        return await _client.PostAsync("/api/GrowthAnalytics/events", content);
    }

    private static GrowthEventRequest CreateEventRequest(
        string eventName, 
        string category, 
        string userId, 
        string sessionId, 
        decimal? eventValue = null)
    {
        return new GrowthEventRequest
        {
            EventName = eventName,
            Category = category,
            UserId = userId,
            SessionId = sessionId,
            ClientTimestamp = DateTime.UtcNow,
            Properties = "{}",
            UtmSource = "test-source",
            UtmMedium = "test-medium",
            UtmCampaign = "test-campaign",
            Referrer = "https://example.com",
            LandingPage = "https://app.example.com/home",
            ScreenResolution = "1920x1080",
            ViewportSize = "1200x800",
            EventValue = eventValue,
            Currency = eventValue.HasValue ? "USD" : null,
            SdkVersion = "1.0.0",
            HasConsent = true,
            ConsentCategories = "analytics,marketing"
        };
    }

    private static bool IsSuccessStatusCode(System.Net.HttpStatusCode statusCode)
    {
        return ((int)statusCode >= 200 && (int)statusCode <= 299);
    }

    private static bool IsExpectedErrorCode(System.Net.HttpStatusCode statusCode)
    {
        // Accept common error codes that indicate the system is working correctly
        return new[] { 400, 401, 403, 404, 422 }.Contains((int)statusCode);
    }
