    #region HTTP Method Tests

    [Theory]
    [InlineData("GET", "/api/content")]
    [InlineData("POST", "/api/auth/login")]
    [InlineData("PUT", "/api/user/profile")]
    [InlineData("DELETE", "/api/watchlist/123")]
    public async Task SecurityHeaders_AppliesHeaders_ToAllHttpMethods(string method, string path)
    {
        // Arrange
        var request = new HttpRequestMessage(new HttpMethod(method), path);

        if (method == "POST")
        {
            request.Content = JsonContent.Create(new { Email = "test@test.com", Password = "Pass123!" });
        }
        else if (method == "PUT")
        {
            request.Content = JsonContent.Create(new { DisplayName = "Test User" });
        }

        // Act
        var response = await Client.SendAsync(request);

        // Assert - Exercises header injection for all methods
        Assert.True(response.Headers.Contains("X-Content-Type-Options"));
        Assert.True(response.Headers.Contains("X-Frame-Options"));
    }

    #endregion

    #region Status Code Tests

    [Theory]
    [InlineData(HttpStatusCode.OK, "/api/content")]
    [InlineData(HttpStatusCode.Unauthorized, "/api/watchlist")]
    [InlineData(HttpStatusCode.NotFound, "/api/content/non-existent")]
    public async Task SecurityHeaders_AppliesHeaders_ToAllStatusCodes(HttpStatusCode expectedStatus, string path)
    {
        // Arrange
        if (expectedStatus == HttpStatusCode.Unauthorized)
        {
            ClearAuthentication();
        }
        else
        {
            SetAuthenticationHeader();
        }

        // Act
        var response = await Client.GetAsync(path);

        // Assert - Exercises header injection for all response codes
        if (response.StatusCode == expectedStatus)
        {
            Assert.True(response.Headers.Contains("X-Content-Type-Options"));
            Assert.True(response.Headers.Contains("X-Frame-Options"));
        }
    }
