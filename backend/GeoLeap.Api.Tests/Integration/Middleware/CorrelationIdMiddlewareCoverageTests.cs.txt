    #region Correlation ID Generation Tests

    [Fact]
    public async Task CorrelationId_GeneratesId_WhenNotProvided()
    {
        // Arrange - No correlation ID in request

        // Act
        var response = await Client.GetAsync("/api/content");

        // Assert - Exercises ID generation path
        Assert.True(response.Headers.Contains("X-Correlation-ID") || response.StatusCode == HttpStatusCode.OK);

        if (response.Headers.Contains("X-Correlation-ID"))
        {
            var correlationId = response.Headers.GetValues("X-Correlation-ID").First();
            Assert.NotEmpty(correlationId);
            Assert.True(Guid.TryParse(correlationId, out _) || !string.IsNullOrEmpty(correlationId));
        }
    }

    [Fact]
    public async Task CorrelationId_PreservesId_WhenProvided()
    {
        // Arrange - Custom correlation ID
        var correlationId = Guid.NewGuid().ToString();
        Client.DefaultRequestHeaders.Add("X-Correlation-ID", correlationId);

        // Act
        var response = await Client.GetAsync("/api/content");

        // Assert - Exercises ID preservation path
        if (response.Headers.Contains("X-Correlation-ID"))
        {
            var responseCorrelationId = response.Headers.GetValues("X-Correlation-ID").First();
            Assert.Equal(correlationId, responseCorrelationId);
        }

        // Cleanup
        Client.DefaultRequestHeaders.Remove("X-Correlation-ID");
    }

    [Fact]
    public async Task CorrelationId_HandlesDifferentIdFormats()
    {
        // Arrange - Different ID formats
        var customIds = new[]
        {
            Guid.NewGuid().ToString(),
            "custom-id-12345",
            "uuid-v4-format",
            "trace-abc-xyz-123"
        };

        foreach (var customId in customIds)
        {
            // Act
            Client.DefaultRequestHeaders.Remove("X-Correlation-ID");
            Client.DefaultRequestHeaders.Add("X-Correlation-ID", customId);

            var response = await Client.GetAsync("/api/content");

            // Assert - Exercises ID format handling
            if (response.Headers.Contains("X-Correlation-ID"))
            {
                var responseId = response.Headers.GetValues("X-Correlation-ID").First();
                Assert.Equal(customId, responseId);
            }
        }

        // Cleanup
        Client.DefaultRequestHeaders.Remove("X-Correlation-ID");
    }

    #region Concurrent Request Tests

    [Fact]
    public async Task CorrelationId_HandlesSimultaneousRequests_WithUniqueIds()
    {
        // Arrange - Multiple simultaneous requests
        var tasks = new List<Task<HttpResponseMessage>>();
        var correlationIds = new List<string>();

        for (int i = 0; i < 10; i++)
        {
            var correlationId = Guid.NewGuid().ToString();
            correlationIds.Add(correlationId);

            var request = new HttpRequestMessage(HttpMethod.Get, "/api/content");
            request.Headers.Add("X-Correlation-ID", correlationId);

            tasks.Add(Client.SendAsync(request));
        }

        // Act
        var responses = await Task.WhenAll(tasks);

        // Assert - Each request maintains its own correlation ID
        for (int i = 0; i < responses.Length; i++)
        {
            if (responses[i].Headers.Contains("X-Correlation-ID"))
            {
                var responseId = responses[i].Headers.GetValues("X-Correlation-ID").First();
                Assert.Equal(correlationIds[i], responseId);
            }
        }
    }

    [Fact]
    public async Task CorrelationId_HandlesSimultaneousRequests_WithoutIds()
    {
        // Arrange - Multiple simultaneous requests without correlation IDs
        var tasks = Enumerable.Range(0, 10)
            .Select(_ => Client.GetAsync("/api/content"))
            .ToList();

        // Act
        var responses = await Task.WhenAll(tasks);

        // Assert - Each request gets a unique generated ID
        var generatedIds = responses
            .Where(r => r.Headers.Contains("X-Correlation-ID"))
            .Select(r => r.Headers.GetValues("X-Correlation-ID").First())
            .ToList();

        if (generatedIds.Any())
        {
            Assert.Equal(generatedIds.Distinct().Count(), generatedIds.Count); // All unique
        }
    }
