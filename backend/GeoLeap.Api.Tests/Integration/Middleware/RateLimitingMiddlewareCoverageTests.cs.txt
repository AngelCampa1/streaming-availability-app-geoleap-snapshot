    #region Rate Limit Path Tests

    [Fact]
    public async Task RateLimiting_WithinLimit_ExecutesAllowPath()
    {
        // Arrange - Normal request rate
        SetAuthenticationHeader();

        // Act - Make requests within limit
        var responses = new List<HttpResponseMessage>();
        for (int i = 0; i < 5; i++)
        {
            responses.Add(await Client.GetAsync("/api/content"));
        }

        // Assert - All requests allowed
        foreach (var response in responses)
        {
            Assert.NotEqual(HttpStatusCode.TooManyRequests, response.StatusCode);

            // Exercises rate limit header injection path
            if (response.Headers.Contains("X-RateLimit-Limit"))
            {
                Assert.True(response.Headers.Contains("X-RateLimit-Remaining"));
                Assert.True(response.Headers.Contains("X-RateLimit-Reset"));
            }
        }
    }

    [Fact]
    public async Task RateLimiting_ExceedingLimit_ExecutesThrottlingPath()
    {
        // Note: Login endpoint has rate limit of 10 requests per 5 minutes
        // This test may not trigger rate limit in test environment if limits are high

        // Arrange
        ClearAuthentication();
        var invalidLogin = new { Email = "test@test.com", Password = "wrong" };

        // Act - Attempt to exceed limit
        var responses = new List<HttpResponseMessage>();
        for (int i = 0; i < 15; i++) // Exceed the 10 request limit
        {
            responses.Add(await Client.PostAsJsonAsync("/api/auth/login", invalidLogin));
        }

        // Assert - Check if any request was rate limited
        var rateLimitedResponse = responses.FirstOrDefault(r => r.StatusCode == HttpStatusCode.TooManyRequests);

        if (rateLimitedResponse != null)
        {
            // Exercises rate limit exceeded path
            Assert.True(rateLimitedResponse.Headers.Contains("Retry-After"));
            Assert.True(rateLimitedResponse.Headers.Contains("X-RateLimit-Limit"));
            Assert.True(rateLimitedResponse.Headers.Contains("X-RateLimit-Remaining"));
            Assert.Equal("0", rateLimitedResponse.Headers.GetValues("X-RateLimit-Remaining").First());

            var errorResponse = await rateLimitedResponse.Content.ReadFromJsonAsync<ApiErrorResponse>();
            Assert.NotNull(errorResponse);
            Assert.Equal("RATE_LIMIT_EXCEEDED", errorResponse.Error.Code);
            Assert.NotNull(errorResponse.Error.RetryAfterSeconds);
        }
    }

    #region Header Injection Path Tests

    [Fact]
    public async Task RateLimiting_InjectsRateLimitHeaders_OnSuccess()
    {
        // Arrange
        SetAuthenticationHeader();

        // Act
        var response = await Client.GetAsync("/api/content");

        // Assert - Exercises header injection for successful requests
        if (response.StatusCode == HttpStatusCode.OK)
        {
            // Rate limit headers may be present
            if (response.Headers.Contains("X-RateLimit-Limit"))
            {
                var limit = int.Parse(response.Headers.GetValues("X-RateLimit-Limit").First());
                Assert.True(limit > 0);

                var remaining = int.Parse(response.Headers.GetValues("X-RateLimit-Remaining").First());
                Assert.True(remaining >= 0);

                var reset = response.Headers.GetValues("X-RateLimit-Reset").First();
                Assert.False(string.IsNullOrEmpty(reset));
            }
        }
    }

    [Fact]
    public async Task RateLimiting_InjectsRetryAfterHeader_OnExceeded()
    {
        // Note: This test may not trigger in all environments due to high limits
        // It verifies the header injection path when rate limit is exceeded

        // Arrange
        ClearAuthentication();
        var invalidLogin = new { Email = "test@test.com", Password = "wrong" };

        // Act - Attempt to exceed limit
        var responses = new List<HttpResponseMessage>();
        for (int i = 0; i < 20; i++)
        {
            responses.Add(await Client.PostAsJsonAsync("/api/auth/login", invalidLogin));
        }

        // Assert
        var rateLimitedResponse = responses.FirstOrDefault(r => r.StatusCode == HttpStatusCode.TooManyRequests);

        if (rateLimitedResponse != null)
        {
            // Exercises Retry-After header injection path
            Assert.True(rateLimitedResponse.Headers.Contains("Retry-After"));
            var retryAfter = rateLimitedResponse.Headers.GetValues("Retry-After").First();
            Assert.True(int.TryParse(retryAfter, out var seconds) && seconds > 0);
        }
    }

    #region Correlation ID Path Tests

    [Fact]
    public async Task RateLimiting_UsesCorrelationIdFromHeaders_WhenProvided()
    {
        // Arrange
        var correlationId = Guid.NewGuid().ToString();
        Client.DefaultRequestHeaders.Add("X-Correlation-ID", correlationId);
        ClearAuthentication();

        // Act - Attempt to exceed limit
        var responses = new List<HttpResponseMessage>();
        for (int i = 0; i < 20; i++)
        {
            responses.Add(await Client.PostAsJsonAsync("/api/auth/login", new
            {
                Email = "test@test.com",
                Password = "wrong"
            }));
        }

        // Assert
        var rateLimitedResponse = responses.FirstOrDefault(r => r.StatusCode == HttpStatusCode.TooManyRequests);

        if (rateLimitedResponse != null)
        {
            var errorResponse = await rateLimitedResponse.Content.ReadFromJsonAsync<ApiErrorResponse>();
            Assert.NotNull(errorResponse);
            Assert.Equal(correlationId, errorResponse.Error.CorrelationId);
        }

        // Cleanup
        Client.DefaultRequestHeaders.Remove("X-Correlation-ID");
    }
