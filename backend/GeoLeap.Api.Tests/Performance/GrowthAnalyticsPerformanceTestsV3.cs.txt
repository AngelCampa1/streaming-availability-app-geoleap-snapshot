    [Fact]
    public async Task BatchEventProcessing_1000Events_CompletesUnder5Seconds()
    {
        // Arrange - High volume batch processing
        const int eventCount = 1000;
        const double maxProcessingTimeSeconds = 5.0;
        
        var events = GenerateTestEvents(eventCount);
        _mockTrackingService.TrackEventsAsync(Arg.Any<IEnumerable<GrowthEvent>>(), Arg.Any<CancellationToken>())
            .Returns(Task.FromResult(eventCount));

        var json = JsonConvert.SerializeObject(events);
        var content = new StringContent(json, Encoding.UTF8, "application/json");

        // Act
        var stopwatch = Stopwatch.StartNew();
        var response = await _client.PostAsync("/api/GrowthAnalytics/events/batch", content);
        stopwatch.Stop();

        // Assert
        _output.WriteLine($"Processed {eventCount} events in {stopwatch.ElapsedMilliseconds}ms");
        Assert.True(stopwatch.TotalSeconds < maxProcessingTimeSeconds, 
            $"Batch processing took {stopwatch.TotalSeconds:F2}s, expected < {maxProcessingTimeSeconds}s");
        
        // Accept success or reasonable error codes
        var acceptableCodes = new[] { 200, 201, 400, 413 }; // 413 = Payload Too Large is acceptable
        Assert.Contains((int)response.StatusCode, acceptableCodes);
    }

    [Theory]
    [InlineData(10, 100)] // 10 concurrent requests, 100ms max each
    [InlineData(50, 200)] // 50 concurrent requests, 200ms max each
    [InlineData(100, 500)] // 100 concurrent requests, 500ms max each
    public async Task ConcurrentEventTracking_MultipleRequests_HandlesLoad(int concurrentRequests, int maxResponseTimeMs)
    {
        // Arrange
        _mockTrackingService.TrackEventAsync(Arg.Any<GrowthEvent>(), Arg.Any<CancellationToken>())
            .Returns(Task.FromResult(true));

        var tasks = new List<Task<(HttpResponseMessage Response, TimeSpan Duration)>>();

        // Act - Create concurrent requests
        for (int i = 0; i < concurrentRequests; i++)
        {
            var eventRequest = CreatePerformanceTestEvent($"concurrent_event_{i}", $"perf_user_{i % 10}");
            tasks.Add(TrackEventWithTiming(eventRequest));
        }

        var results = await Task.WhenAll(tasks);

        // Assert
        var successfulRequests = results.Where(r => IsSuccessOrAcceptableError(r.Response.StatusCode)).ToList();
        var averageResponseTime = successfulRequests.Average(r => r.Duration.TotalMilliseconds);
        var maxResponseTime = successfulRequests.Max(r => r.Duration.TotalMilliseconds);

        _output.WriteLine($"Concurrent requests: {concurrentRequests}");
        _output.WriteLine($"Successful: {successfulRequests.Count}/{concurrentRequests}");
        _output.WriteLine($"Average response time: {averageResponseTime:F2}ms");
        _output.WriteLine($"Max response time: {maxResponseTime:F2}ms");

        // Performance assertions
        Assert.True(successfulRequests.Count >= concurrentRequests * 0.8, // At least 80% success rate
            $"Success rate too low: {successfulRequests.Count}/{concurrentRequests}");
        
        Assert.True(averageResponseTime < maxResponseTimeMs, 
            $"Average response time {averageResponseTime:F2}ms exceeds limit {maxResponseTimeMs}ms");
    }

    [Fact]
    public async Task MemoryUsage_LargePayloads_RemainsStable()
    {
        // Arrange - Memory usage test with large payloads
        const int largeEventCount = 500;
        var initialMemory = GC.GetTotalMemory(false);

        // Act - Process multiple large batches
        for (int batch = 0; batch < 5; batch++)
        {
            var events = GenerateTestEvents(largeEventCount);
            _mockTrackingService.TrackEventsAsync(Arg.Any<IEnumerable<GrowthEvent>>(), Arg.Any<CancellationToken>())
                .Returns(Task.FromResult(largeEventCount));

            var json = JsonConvert.SerializeObject(events);
            var content = new StringContent(json, Encoding.UTF8, "application/json");
            
            var response = await _client.PostAsync("/api/GrowthAnalytics/events/batch", content);
            
            // Force garbage collection after each batch
            GC.Collect();
            GC.WaitForPendingFinalizers();
            GC.Collect();
        }

        var finalMemory = GC.GetTotalMemory(false);
        var memoryIncrease = finalMemory - initialMemory;
        var memoryIncreaseKB = memoryIncrease / 1024;

        // Assert
        _output.WriteLine($"Memory usage - Initial: {initialMemory / 1024}KB, Final: {finalMemory / 1024}KB, Increase: {memoryIncreaseKB}KB");
        
        // Memory increase should be reasonable (less than 10MB for this test)
        Assert.True(memoryIncrease < 10 * 1024 * 1024, 
            $"Memory usage increased by {memoryIncreaseKB}KB, which seems excessive");
    }

    private async Task<(HttpResponseMessage Response, TimeSpan Duration)> TrackEventWithTiming(GrowthEventRequest eventRequest)
    {
        var stopwatch = Stopwatch.StartNew();
        var json = JsonConvert.SerializeObject(eventRequest);
        var content = new StringContent(json, Encoding.UTF8, "application/json");
        var response = await _client.PostAsync("/api/GrowthAnalytics/events", content);
        stopwatch.Stop();
        
        return (response, stopwatch.Elapsed);
    }

    private static List<GrowthEventRequest> GenerateTestEvents(int count)
    {
        return Enumerable.Range(1, count)
            .Select(i => CreatePerformanceTestEvent($"perf_event_{i}", $"perf_user_{i % 100}"))
            .ToList();
    }

    private static GrowthEventRequest CreatePerformanceTestEvent(string eventName, string userId)
    {
        return new GrowthEventRequest
        {
            EventName = eventName,
            Category = "performance_test",
            UserId = userId,
            SessionId = $"perf_session_{Guid.NewGuid():N}",
            ClientTimestamp = DateTime.UtcNow,
            Properties = "{\"test\": true}",
            UtmSource = "performance",
            UtmMedium = "test",
            UtmCampaign = "load_test",
            HasConsent = true,
            ConsentCategories = "analytics"
        };
    }

    private static List<AttributionTouch> GenerateTestTouchpoints(int count)
    {
        var channels = new[] { "organic_search", "paid_search", "email", "social", "direct", "referral" };
        
        return Enumerable.Range(1, count)
            .Select(i => new AttributionTouch
            {
                Id = Guid.NewGuid(),
                EventName = $"touchpoint_{i}",
                Channel = channels[i % channels.Length],
                Source = $"source_{i}",
                Medium = $"medium_{i}",
                Timestamp = DateTime.UtcNow.AddDays(-i),
                Attribution = 1.0m / count,
                Position = i,
                PositionFromEnd = count - i + 1
            }).ToList();
    }

            .Select(i => new ChannelComparisonResult

                ModelResults = new List<ChannelModelResult>(),
                VariancePercentage = 10.0m + (i * 2.0m),

            }).ToList();
    }

    private static bool IsSuccessOrAcceptableError(System.Net.HttpStatusCode statusCode)
    {
        var code = (int)statusCode;
        return (code >= 200 && code <= 299) || // Success codes
               new[] { 400, 401, 403, 404, 413, 422 }.Contains(code); // Acceptable error codes
    }

    private void SetupPerformanceMocks()
    {
        // Configure mocks to simulate realistic performance characteristics
        _mockTrackingService.TrackEventAsync(Arg.Any<GrowthEvent>(), Arg.Any<CancellationToken>())
            .Returns(async call =>
            {
                // Simulate small processing delay
                await Task.Delay(1);
                return true;
            });

        _mockTrackingService.TrackEventsAsync(Arg.Any<IEnumerable<GrowthEvent>>(), Arg.Any<CancellationToken>())
            .Returns(async call =>
            {
                var events = call.Arg<IEnumerable<GrowthEvent>>().ToList();
                // Simulate batch processing delay proportional to batch size
                await Task.Delay(Math.Min(events.Count / 10, 100));
                return events.Count;
            });

        _mockAttributionService.CalculateAttributionAsync(Arg.Any<Guid>(), Arg.Any<Guid?>())
            .Returns(async call =>
            {
                // Simulate attribution calculation delay
                await Task.Delay(5);
                return new AttributionResult
                {
                    ConversionEventId = call.Arg<Guid>(),
                    TotalAttribution = 1.0m,
                    Touchpoints = GenerateTestTouchpoints(3)
                };
            });

        _mockTrackingService.GetProcessingStatsAsync()
            .Returns(async call =>
            {
                // Simulate stats retrieval delay
                await Task.Delay(2);
                return GenerateProcessingStats();
            });
    }
