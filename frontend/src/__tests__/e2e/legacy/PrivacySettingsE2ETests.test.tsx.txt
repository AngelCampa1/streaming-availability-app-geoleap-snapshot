  test('Privacy Settings Accessibility Compliance', async () => {
    await page.goto(`${BASE_URL}/profile/privacy`);
    
    // Test keyboard navigation
    await page.keyboard.press('Tab');
    let focusedElement = page.locator(':focus');
    await expect(focusedElement).toBeVisible();
    
    // Test through multiple controls
    for (let i = 0; i < 5; i++) {
      await page.keyboard.press('Tab');
      focusedElement = page.locator(':focus');
      await expect(focusedElement).toBeVisible();
    }
    
    // Test ARIA attributes
    const toggles = page.locator('[role="switch"]');
    const toggleCount = await toggles.count();
    expect(toggleCount).toBeGreaterThan(0);
    
    for (let i = 0; i < Math.min(toggleCount, 3); i++) {
      const toggle = toggles.nth(i);
      await expect(toggle).toHaveAttribute('aria-checked');
      await expect(toggle).toHaveAttribute('aria-labelledby');
    }
    
    // Test screen reader labels
    const sections = page.locator('[data-testid$="-section"]');
    const sectionCount = await sections.count();
    
    for (let i = 0; i < Math.min(sectionCount, 3); i++) {
      const section = sections.nth(i);
      await expect(section).toHaveAttribute('aria-labelledby');
    }
    
    // Test high contrast mode
    await page.addStyleTag({
      content: `
        @media (prefers-contrast: high) {
          * { filter: contrast(150%); }
        }
      `
    });
