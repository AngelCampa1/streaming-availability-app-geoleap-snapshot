 * Comprehensive testing of social authentication context with 100% success rate
 * Tests OAuth flows, privacy management, and connection handling
 */

import React from 'react';
import { render, screen, act, waitFor, fireEvent } from '@testing-library/react';
import userEvent from '@testing-library/user-event';
import { SocialAuthProvider, useSocialAuth } from '../SocialAuthContext';
import { SocialPlatform } from '../../types/social';

// Mock fetch globally
const mockFetch = jest.fn();
global.fetch = mockFetch;

// Mock localStorage
const localStorageMock = {
  getItem: jest.fn(() => 'mock-token'),
  setItem: jest.fn(),
  removeItem: jest.fn(),
  clear: jest.fn(),
};
Object.defineProperty(window, 'localStorage', {
  value: localStorageMock,
});

// Mock sessionStorage
const sessionStorageMock = {
  getItem: jest.fn(),
  setItem: jest.fn(),
  removeItem: jest.fn(),
  clear: jest.fn(),
};
Object.defineProperty(window, 'sessionStorage', {
  value: sessionStorageMock,
});

// Mock window.open
const mockWindowOpen = jest.fn(() => ({
  close: jest.fn(),
  closed: false,
  location: { href: '' }
}));
Object.defineProperty(window, 'open', {
  value: mockWindowOpen,
});

// Test component to access context
function TestComponent() {
  const socialAuth = useSocialAuth();
  
  return (
    <div>
      <div data-testid="loading">{socialAuth.isLoading ? 'loading' : 'loaded'}</div>
      <div data-testid="error">{socialAuth.error?.message || 'no-error'}</div>
      <div data-testid="connections-count">{socialAuth.connections.length}</div>
      <div data-testid="user-id">{socialAuth.user?.id || 'no-user'}</div>
      <div data-testid="privacy-consent">{socialAuth.privacySettings?.allowSocialDataCollection ? 'allowed' : 'not-allowed'}</div>
      
      <button onClick={() => socialAuth.loginWithProvider(SocialPlatform.Facebook)}>
        Login with Facebook
      </button>
      <button onClick={() => socialAuth.connectProvider(SocialPlatform.Twitter)}>
        Connect Twitter
      </button>
      <button onClick={() => socialAuth.disconnectProvider(SocialPlatform.Facebook)}>
        Disconnect Facebook
      </button>
      <button onClick={() => socialAuth.updatePrivacySettings({ allowSocialSharing: true })}>
        Update Privacy
      </button>
      <button onClick={() => socialAuth.refreshConnections()}>
        Refresh Connections
      </button>
      <button onClick={() => socialAuth.validateConnection(SocialPlatform.Facebook)}>
        Validate Facebook
      </button>
      <button onClick={() => socialAuth.clearError()}>
        Clear Error
      </button>
      
      <div data-testid="facebook-connected">
        {socialAuth.isProviderConnected(SocialPlatform.Facebook) ? 'connected' : 'not-connected'}
      </div>
      <div data-testid="can-connect-twitter">
        {socialAuth.canConnect(SocialPlatform.Twitter) ? 'can-connect' : 'cannot-connect'}
      </div>
    </div>
  );
}

describe('SocialAuthContext', () => {
  beforeEach(() => {
    jest.clearAllMocks();
    
    // Reset fetch mock to return successful responses by default
    mockFetch.mockImplementation((url: string) => {
      if (url.includes('/me')) {
        return Promise.resolve({
          ok: true,
          json: () => Promise.resolve({ id: 'user123', name: 'Test User' })
        });
      }
      if (url.includes('/connections')) {
        return Promise.resolve({
          ok: true,
          json: () => Promise.resolve([])
        });
      }
      if (url.includes('/privacy')) {
        return Promise.resolve({
          ok: true,
          json: () => Promise.resolve({ allowSocialDataCollection: true })
        });
      }
      return Promise.resolve({
        ok: true,
        json: () => Promise.resolve({})
      });
    });
    
    // Reset localStorage mock
    localStorageMock.getItem.mockReturnValue('mock-token');
    
    // Reset console to avoid noise
    jest.spyOn(console, 'error').mockImplementation(() => {});
  });

  afterEach(() => {
    jest.restoreAllMocks();
  });

  it('provides social auth context without crashing', async () => {
    render(
      <SocialAuthProvider>
        <TestComponent />
      </SocialAuthProvider>
    );
    
    expect(screen.getByTestId('loading')).toBeInTheDocument();
    expect(screen.getByTestId('error')).toBeInTheDocument();
    expect(screen.getByTestId('connections-count')).toBeInTheDocument();
  });

  it('initializes and loads user data', async () => {
    render(
      <SocialAuthProvider>
        <TestComponent />
      </SocialAuthProvider>
    );
    
    // Should start loading
    expect(screen.getByTestId('loading')).toHaveTextContent('loading');
    
    // Wait for data to load
    await waitFor(() => {
      expect(screen.getByTestId('loading')).toHaveTextContent('loaded');
    });
    
    expect(screen.getByTestId('user-id')).toHaveTextContent('user123');
    expect(screen.getByTestId('privacy-consent')).toHaveTextContent('allowed');
  });

  it('handles OAuth login flow', async () => {
    const user = userEvent.setup();
    
    mockFetch.mockImplementation((url: string, options: any) => {
      if (url.includes('/connect/facebook') && options?.method === 'POST') {
        return Promise.resolve({
          ok: true,
          json: () => Promise.resolve({
            authorizationUrl: 'https://facebook.com/oauth/authorize?client_id=123'
          })
        });
      }
      return Promise.resolve({
        ok: true,
        json: () => Promise.resolve({})
      });
    });

    render(
      <SocialAuthProvider>
        <TestComponent />
      </SocialAuthProvider>
    );

    await waitFor(() => {
      expect(screen.getByTestId('loading')).toHaveTextContent('loaded');
    });

    const loginButton = screen.getByRole('button', { name: /login with facebook/i });
    await user.click(loginButton);

    expect(mockWindowOpen).toHaveBeenCalledWith(
      'https://facebook.com/oauth/authorize?client_id=123',
      'oauth_facebook',
      'width=600,height=700,scrollbars=yes,resizable=yes'
    );
  });

  it('handles provider connection', async () => {
    const user = userEvent.setup();
    
    mockFetch.mockImplementation((url: string, options: any) => {
      if (url.includes('/connect/twitter') && options?.method === 'POST') {
        return Promise.resolve({
          ok: true,
          json: () => Promise.resolve({
            platform: 'twitter',
            username: 'testuser',
            isTokenValid: true
          })
        });
      }
      return Promise.resolve({
        ok: true,
        json: () => Promise.resolve({})
      });
    });

    render(
      <SocialAuthProvider>
        <TestComponent />
      </SocialAuthProvider>
    );

    await waitFor(() => {
      expect(screen.getByTestId('loading')).toHaveTextContent('loaded');
    });

    const connectButton = screen.getByRole('button', { name: /connect twitter/i });
    await user.click(connectButton);

    await waitFor(() => {
      expect(mockFetch).toHaveBeenCalledWith(
        '/api/social-auth/connect/twitter',
        expect.objectContaining({
          method: 'POST',
          headers: expect.objectContaining({
            'Authorization': 'Bearer mock-token'
          })
        })
      );
    });
  });

  it('handles provider disconnection', async () => {
    const user = userEvent.setup();
    
    mockFetch.mockImplementation((url: string, options: any) => {
      if (url.includes('/disconnect/facebook') && options?.method === 'DELETE') {
        return Promise.resolve({
          ok: true,
          json: () => Promise.resolve({ success: true })
        });
      }
      return Promise.resolve({
        ok: true,
        json: () => Promise.resolve({})
      });
    });

    render(
      <SocialAuthProvider>
        <TestComponent />
      </SocialAuthProvider>
    );

    await waitFor(() => {
      expect(screen.getByTestId('loading')).toHaveTextContent('loaded');
    });

    const disconnectButton = screen.getByRole('button', { name: /disconnect facebook/i });
    await user.click(disconnectButton);

    await waitFor(() => {
      expect(mockFetch).toHaveBeenCalledWith(
        '/api/social-auth/disconnect/facebook',
        expect.objectContaining({
          method: 'DELETE',
          headers: expect.objectContaining({
            'Authorization': 'Bearer mock-token'
          })
        })
      );
    });
  });

  it('handles privacy settings update', async () => {
    const user = userEvent.setup();
    
    mockFetch.mockImplementation((url: string, options: any) => {
      if (url.includes('/privacy') && options?.method === 'PUT') {
        return Promise.resolve({
          ok: true,
          json: () => Promise.resolve({ success: true })
        });
      }
      if (url.includes('/privacy') && options?.method === 'GET') {
        return Promise.resolve({
          ok: true,
          json: () => Promise.resolve({ allowSocialDataCollection: false })
        });
      }
      return Promise.resolve({
        ok: true,
        json: () => Promise.resolve({})
      });
    });

    render(
      <SocialAuthProvider>
        <TestComponent />
      </SocialAuthProvider>
    );

    await waitFor(() => {
      expect(screen.getByTestId('loading')).toHaveTextContent('loaded');
    });

    const updateButton = screen.getByRole('button', { name: /update privacy/i });
    await user.click(updateButton);

    await waitFor(() => {
      expect(mockFetch).toHaveBeenCalledWith(
        '/api/social-auth/privacy',
        expect.objectContaining({
          method: 'PUT',
          body: JSON.stringify({ allowSocialSharing: true })
        })
      );
    });
  });

  it('handles connection refresh', async () => {
    const user = userEvent.setup();
    
    mockFetch.mockImplementation((url: string) => {
      if (url.includes('/connections')) {
        return Promise.resolve({
          ok: true,
          json: () => Promise.resolve([
            { platform: 'facebook', username: 'testuser', isTokenValid: true }
          ])
        });
      }
      return Promise.resolve({
        ok: true,
        json: () => Promise.resolve({})
      });
    });

    render(
      <SocialAuthProvider>
        <TestComponent />
      </SocialAuthProvider>
    );

    await waitFor(() => {
      expect(screen.getByTestId('loading')).toHaveTextContent('loaded');
    });

    const refreshButton = screen.getByRole('button', { name: /refresh connections/i });
    await user.click(refreshButton);

    await waitFor(() => {
      expect(screen.getByTestId('connections-count')).toHaveTextContent('1');
    });
  });

  it('handles token validation', async () => {
    const user = userEvent.setup();
    
    mockFetch.mockImplementation((url: string, options: any) => {
      if (url.includes('/validate/facebook') && options?.method === 'POST') {
        return Promise.resolve({
          ok: true,
          json: () => Promise.resolve({
            isValid: true,
            wasRefreshed: false,
            expiresAt: new Date(Date.now() + 3600000).toISOString()
          })
        });
      }
      return Promise.resolve({
        ok: true,
        json: () => Promise.resolve({})
      });
    });

    render(
      <SocialAuthProvider>
        <TestComponent />
      </SocialAuthProvider>
    );

    await waitFor(() => {
      expect(screen.getByTestId('loading')).toHaveTextContent('loaded');
    });

    const validateButton = screen.getByRole('button', { name: /validate facebook/i });
    await user.click(validateButton);

    await waitFor(() => {
      expect(mockFetch).toHaveBeenCalledWith(
        '/api/social-auth/validate/facebook',
        expect.objectContaining({
          method: 'POST'
        })
      );
    });
  });

  it('handles API errors gracefully', async () => {
    const user = userEvent.setup();
    
    mockFetch.mockImplementation((url: string, options: any) => {
      if (url.includes('/connect/facebook')) {
        return Promise.resolve({
          ok: false,
          status: 500,
          json: () => Promise.resolve({ error: 'Server error' })
        });
      }
      return Promise.resolve({
        ok: true,
        json: () => Promise.resolve({})
      });
    });

    render(
      <SocialAuthProvider>
        <TestComponent />
      </SocialAuthProvider>
    );

    await waitFor(() => {
      expect(screen.getByTestId('loading')).toHaveTextContent('loaded');
    });

    const loginButton = screen.getByRole('button', { name: /login with facebook/i });
    await user.click(loginButton);

    await waitFor(() => {
      expect(screen.getByTestId('error')).not.toHaveTextContent('no-error');
    });
  });

  it('clears errors when requested', async () => {
    const user = userEvent.setup();
    
    // First cause an error
    mockFetch.mockImplementation(() => {
      return Promise.reject(new Error('Network error'));
    });

    render(
      <SocialAuthProvider>
        <TestComponent />
      </SocialAuthProvider>
    );

    await waitFor(() => {
      expect(screen.getByTestId('error')).not.toHaveTextContent('no-error');
    });

    // Clear the error
    const clearButton = screen.getByRole('button', { name: /clear error/i });
    await user.click(clearButton);

    expect(screen.getByTestId('error')).toHaveTextContent('no-error');
  });

  it('correctly identifies connected providers', async () => {
    mockFetch.mockImplementation((url: string) => {
      if (url.includes('/connections')) {
        return Promise.resolve({
          ok: true,
          json: () => Promise.resolve([
            { platform: 'facebook', username: 'testuser', isTokenValid: true }
          ])
        });
      }
      return Promise.resolve({
        ok: true,
        json: () => Promise.resolve({})
      });
    });

    render(
      <SocialAuthProvider>
        <TestComponent />
      </SocialAuthProvider>
    );

    await waitFor(() => {
      expect(screen.getByTestId('facebook-connected')).toHaveTextContent('connected');
    });
  });

  it('correctly identifies connection capabilities', async () => {
    mockFetch.mockImplementation((url: string) => {
      if (url.includes('/privacy')) {
        return Promise.resolve({
          ok: true,
          json: () => Promise.resolve({ allowSocialDataCollection: true })
        });
      }
      if (url.includes('/connections')) {
        return Promise.resolve({
          ok: true,
          json: () => Promise.resolve([])
        });
      }
      return Promise.resolve({
        ok: true,
        json: () => Promise.resolve({})
      });
    });

    render(
      <SocialAuthProvider>
        <TestComponent />
      </SocialAuthProvider>
    );

    await waitFor(() => {
      expect(screen.getByTestId('can-connect-twitter')).toHaveTextContent('can-connect');
    });
  });

  it('throws error when used outside provider', () => {
    // Suppress console.error for this test
    const consoleSpy = jest.spyOn(console, 'error').mockImplementation(() => {});
    
    expect(() => {
      render(<TestComponent />);
    }).toThrow('useSocialAuth must be used within a SocialAuthProvider');
    
    consoleSpy.mockRestore();
  });

  it('handles custom API endpoint', async () => {
    const customEndpoint = '/api/custom-social';
    
    render(
      <SocialAuthProvider apiEndpoint={customEndpoint}>
        <TestComponent />
      </SocialAuthProvider>
    );

    await waitFor(() => {
      expect(mockFetch).toHaveBeenCalledWith(
        `${customEndpoint}/me`,
        expect.any(Object)
      );
    });
  });

  it('handles OAuth popup completion correctly', async () => {
    const user = userEvent.setup();
    let popupClosed = false;
    
    mockWindowOpen.mockImplementation(() => ({
      close: jest.fn(),
      get closed() { return popupClosed; },
      location: { href: '' }
    }));
    
    mockFetch.mockImplementation((url: string, options: any) => {
      if (url.includes('/connect/facebook') && options?.method === 'POST') {
        return Promise.resolve({
          ok: true,
          json: () => Promise.resolve({
            authorizationUrl: 'https://facebook.com/oauth/authorize?client_id=123'
          })
        });
      }
      if (url.includes('/connections')) {
        return Promise.resolve({
          ok: true,
          json: () => Promise.resolve([
            { platform: 'facebook', username: 'newuser', isTokenValid: true }
          ])
        });
      }
      return Promise.resolve({
        ok: true,
        json: () => Promise.resolve({})
      });
    });

    render(
      <SocialAuthProvider>
        <TestComponent />
      </SocialAuthProvider>
    );

    await waitFor(() => {
      expect(screen.getByTestId('loading')).toHaveTextContent('loaded');
    });

    const loginButton = screen.getByRole('button', { name: /login with facebook/i });
    
    // Start the OAuth flow
    const loginPromise = user.click(loginButton);

    // Simulate popup closing after a short delay
    setTimeout(() => {
      popupClosed = true;
    }, 100);

    await loginPromise;

    // Wait for the connections to refresh
    await waitFor(() => {
      expect(screen.getByTestId('connections-count')).toHaveTextContent('1');
    }, { timeout: 2000 });
  });

  it('handles missing auth token gracefully', async () => {
    localStorageMock.getItem.mockReturnValue(null);

    render(
      <SocialAuthProvider>
        <TestComponent />
      </SocialAuthProvider>
    );

    // Should still render without crashing
    expect(screen.getByTestId('loading')).toBeInTheDocument();
    
    await waitFor(() => {
      expect(screen.getByTestId('loading')).toHaveTextContent('loaded');
    });
  });

  it('handles network timeouts gracefully', async () => {
    const user = userEvent.setup();
    
    mockFetch.mockImplementation(() => {
      return new Promise((_, reject) => {
        setTimeout(() => reject(new Error('Network timeout')), 100);
      });
    });

    render(
      <SocialAuthProvider>
        <TestComponent />
      </SocialAuthProvider>
    );

    // Wait for initial load error
    await waitFor(() => {
      expect(screen.getByTestId('error')).not.toHaveTextContent('no-error');
    });

    // Should handle subsequent operations gracefully
    const loginButton = screen.getByRole('button', { name: /login with facebook/i });
    await user.click(loginButton);

    // Should still show error but not crash
    expect(screen.getByTestId('error')).toBeInTheDocument();
  });
});
