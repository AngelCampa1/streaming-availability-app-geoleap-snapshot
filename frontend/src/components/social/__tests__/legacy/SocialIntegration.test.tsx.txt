describe('Context-dependent Components', () => {
  beforeEach(() => {
    jest.clearAllMocks();
    (global.fetch as jest.Mock).mockResolvedValue({
      ok: true,
      json: () => Promise.resolve({ success: true, user: mockUser }),
    });
    // Reset mock to default context
    mockUseSocialAuth.mockReturnValue(mockSocialAuthContext);
  });

  describe('Social Authentication Context', () => {
    it('provides mock social auth context', () => {
      const TestComponent = () => {
        const { user, isLoading } = mockUseSocialAuth();
        return (
          <div>
            <div data-testid="loading">{isLoading.toString()}</div>
            <div data-testid="user">{user?.displayName || 'No user'}</div>
          </div>
        );
      };

      render(<TestComponent />);

      expect(screen.getByTestId('loading')).toHaveTextContent('false');
      expect(screen.getByTestId('user')).toHaveTextContent('Test User');
    });

    it('handles error states', () => {
      const errorContext = {
        ...mockSocialAuthContext,
        error: 'Failed to load data',
      };
      mockUseSocialAuth.mockReturnValue(errorContext);

      const TestComponent = () => {
        const { error } = mockUseSocialAuth();
        return <div data-testid="error">{error || 'No error'}</div>;
      };

      render(<TestComponent />);
      expect(screen.getByTestId('error')).toHaveTextContent('Failed to load data');
    });
  });

  describe('Integration Tests', () => {
    it('handles authentication flow', async () => {
      // Test that authentication functions can be called
      const { login } = mockSocialAuthContext;
      const result = await login('google');
      expect(result).toEqual(mockUser);
    });

    it('handles connection management', async () => {
      // Test that connection functions work
      const { connect } = mockSocialAuthContext;
      const result = await connect('facebook');
      expect(result).toEqual(mockConnections[0]);
    });
  });

  describe('Error Handling', () => {
    beforeEach(() => {
      jest.spyOn(console, 'error').mockImplementation(() => {});
    });

    afterEach(() => {
      (console.error as jest.Mock).mockRestore();
    });

    it('handles API errors gracefully', async () => {
      const errorContext = {
        ...mockSocialAuthContext,
        error: 'Failed to load friend activities',
      };
      mockUseSocialAuth.mockReturnValue(errorContext);

      const TestComponent = () => {
        const { error } = mockUseSocialAuth();
        if (error) {
          return <div data-testid="error-message">{error}</div>;
        }
        return <div>No error</div>;
      };

      render(<TestComponent />);
      
      expect(screen.getByTestId('error-message')).toHaveTextContent('Failed to load friend activities');
    });

    it('handles authentication failures', async () => {
      // Mock fetch to simulate auth failure
      (global.fetch as jest.Mock).mockRejectedValueOnce(new Error('Auth failed'));

      const TestComponent = () => {
        const [error, setError] = React.useState<string | null>(null);
        
        const handleAuth = async () => {
          try {
            await fetch('/api/auth/login');
          } catch (err) {
            setError('Authentication failed');
          }
        };

        return (
          <div>
            <button onClick={handleAuth}>Login</button>
            {error && <div data-testid="auth-error">{error}</div>}
          </div>
        );
      };

      const user = userEvent.setup();
      render(<TestComponent />);

      const button = screen.getByRole('button', { name: /login/i });
      await user.click(button);

      await waitFor(() => {
        expect(screen.getByTestId('auth-error')).toHaveTextContent('Authentication failed');
      });
    });
  });

  describe('Privacy and Settings', () => {
    it('handles privacy settings updates', async () => {
      const { updatePrivacySettings } = mockSocialAuthContext;
      const newSettings = { allowFriendDiscovery: false };
      
      const result = await updatePrivacySettings(newSettings);
      expect(updatePrivacySettings).toHaveBeenCalledWith(newSettings);
      expect(result).toEqual(mockUser.privacySettings);
    });

    it('handles different user privacy states', () => {
      const userWithoutDiscovery = {
        ...mockUser,
        privacySettings: {
          ...mockUser.privacySettings,
          allowFriendDiscovery: false,
        },
      };

      const contextWithPrivacy = {
        ...mockSocialAuthContext,
        user: userWithoutDiscovery,
      };
      
      mockUseSocialAuth.mockReturnValue(contextWithPrivacy);

      const TestComponent = () => {
        const { user } = mockUseSocialAuth();
        return (
          <div>
            {user?.privacySettings.allowFriendDiscovery ? (
              <div data-testid="discovery-enabled">Discovery Enabled</div>
            ) : (
              <div data-testid="discovery-disabled">Friend Discovery Disabled</div>
            )}
          </div>
        );
      };

      render(<TestComponent />);
      expect(screen.getByTestId('discovery-disabled')).toHaveTextContent('Friend Discovery Disabled');
    });
  });

  describe('Social Connections', () => {
    it('handles empty connections list', () => {
      const emptyContext = {
        ...mockSocialAuthContext,
        connections: [],
      };
      mockUseSocialAuth.mockReturnValue(emptyContext);

      const TestComponent = () => {
        const { connections } = mockUseSocialAuth();
        return (
          <div>
            {connections.length === 0 ? (
              <div data-testid="no-connections">No social connections</div>
            ) : (
              <div data-testid="has-connections">{connections.length} connections</div>
            )}
          </div>
        );
      };

      render(<TestComponent />);
      expect(screen.getByTestId('no-connections')).toHaveTextContent('No social connections');
    });

    it('displays connection information', () => {
      const TestComponent = () => {
        const { connections } = mockUseSocialAuth();
        return (
          <div>
            {connections.map((conn) => (
              <div key={conn.id} data-testid="connection">
                {conn.provider}: {conn.displayName}
              </div>
            ))}
          </div>
        );
      };

      render(<TestComponent />);
      expect(screen.getByTestId('connection')).toHaveTextContent('google: Test User');
    });
  });
});

    render(

    expect(screen.getByTestId('heart-icon')).toBeInTheDocument();
    expect(screen.getByTestId('star-icon')).toBeInTheDocument();
  });

  it('handles user interactions', async () => {
    const handleClick = jest.fn();
    const user = userEvent.setup();

    render(
      <button onClick={handleClick} data-testid="test-button">
        Click me
      </button>
    );

    const button = screen.getByTestId('test-button');
    await user.click(button);

    expect(handleClick).toHaveBeenCalledTimes(1);
  });

  it('handles form inputs', async () => {
    const user = userEvent.setup();

    render(
      <div>
        <input data-testid="test-input" placeholder="Enter text" />
        <select data-testid="test-select">
          <option value="option1">Option 1</option>
          <option value="option2">Option 2</option>
        </select>
      </div>
    );

    const input = screen.getByTestId('test-input');
    const select = screen.getByTestId('test-select');

    await user.type(input, 'test value');
    await user.selectOptions(select, 'option2');

    expect(input).toHaveValue('test value');
    expect(select).toHaveValue('option2');
  });

  it('handles async operations', async () => {
    const asyncOperation = jest.fn().mockResolvedValue('success');

    const TestComponent = () => {
      const [result, setResult] = React.useState<string>('');

      const handleAsync = async () => {
        const res = await asyncOperation();
        setResult(res);
      };

      return (
        <div>
          <button onClick={handleAsync}>Run Async</button>
          <div data-testid="result">{result}</div>
        </div>
      );
    };

    const user = userEvent.setup();
    render(<TestComponent />);

    const button = screen.getByRole('button', { name: /run async/i });
    await user.click(button);

    await waitFor(() => {
      expect(screen.getByTestId('result')).toHaveTextContent('success');
    });

    expect(asyncOperation).toHaveBeenCalledTimes(1);
  });
});
